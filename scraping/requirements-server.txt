# Server-only runtime dependencies for the VideoGameWizard RAG server.
#
# A strict subset of requirements.txt: the FastAPI app plus the embedding /
# vector-store stack — WITHOUT the scraping pipeline (scrapy / beautifulsoup4 /
# requests). The Docker image installs only this file so the container stays
# slim: it serves queries, it never (re)builds the dataset.
#
# For test/lint-only deps see requirements-test.txt.
#
# REPRODUCIBILITY: this file holds intent (floors). The Docker image installs
# the fully-pinned requirements-server.lock (compiled on Linux/py3.12 with
# pip-compile — regeneration command is in the lock's header). Update flow:
# edit the floors here, re-run pip-compile, commit both files.

# --- RAG server ---
fastapi>=0.139.0
uvicorn[standard]>=0.30
pydantic>=2.9
pydantic-settings>=2.4
httpx>=0.27

# --- Embeddings + vector store (imported lazily at server startup) ---
sentence-transformers>=3.0
chromadb>=1.5.9

# PyTorch: NOTE — on Linux a plain `pip install torch` from PyPI is the CUDA
# build (~5-7 GB with its nvidia-* wheels). The Dockerfile pre-installs the
# CPU-only wheel (--index-url https://download.pytorch.org/whl/cpu) before this
# file so the container stays slim. For GPU embedding on a host, use the CUDA
# build instead, e.g.
#   pip install torch --index-url https://download.pytorch.org/whl/cu128
torch>=2.3
